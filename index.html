async function askWorldAI() { const query = document.getElementById('aiInp').value; const resBox = document.getElementById('aiResponse'); if(!query) return alert("দয়া করে বিষয়টি লিখুন!"); resBox.style.display = "block"; resBox.innerHTML = "" + query + " সম্পর্কে বাংলা ও ইংরেজিতে বিস্তারিত উত্তর তৈরি করছি..."; try { // বাংলা উইকিপিডিয়া থেকে তথ্য খোঁজা const searchResponse = await fetch(`https://bn.wikipedia.org/w/api.php?action=query&format=json&prop=extracts&exintro=false&explaintext=true&titles=${encodeURIComponent(query)}&origin=*`); const searchData = await searchResponse.json(); const pages = searchData.query.pages; const pageId = Object.keys(pages)[0]; let contentBN = pages[pageId].extract; if (contentBN && pageId !== "-1") { // উত্তরটি অনেক বড় হলে প্রথম অংশটুকু নেওয়া হচ্ছে contentBN = contentBN.split('.').slice(0, 10).join('.') + '.'; resBox.innerHTML = `

${query} (In Bengali)

${contentBN}

English Translation (ইংরেজিতে)

অনুবাদ করা হচ্ছে... (Translating to English)
`; // ইংরেজি অনুবাদের জন্য গুগল ট্রান্সলেট ফ্রি API (একটি ছোট ট্রিক) const transUrl = `https://translate.googleapis.com/translate_a/single?client=gtx&sl=bn&tl=en&dt=t&q=${encodeURIComponent(contentBN)}`; const transRes = await fetch(transUrl); const transData = await transRes.json(); const contentEN = transData[0].map(x => x[0]).join(''); document.getElementById('englishContent').innerText = contentEN; } else { resBox.innerHTML = "দুঃখিত, এই বিষয়টি সম্পর্কে বিস্তারিত তথ্য পাওয়া যায়নি। সঠিক বানান লিখুন।"; } } catch (error) { resBox.innerHTML = "ইন্টারনেট সংযোগ চেক করুন।"; } document.getElementById('aiInp').value = ""; }
